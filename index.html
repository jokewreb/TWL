async function start() { try { const res = await fetch('./levels.json'); let data = await res.json(); // Ensure points are sorted correctly (Highest at Top) data.sort((a, b) => b.points - a.points); const sidebar = document.getElementById('sidebar-list'); sidebar.innerHTML = ''; // Clear loading text data.forEach((lvl, i) => { const rank = i + 1; const item = document.createElement('div'); // Apply rank-specific colors (Gold, Silver, Bronze) let rankClass = "rank-default"; if(rank === 1) rankClass = "rank-1"; if(rank === 2) rankClass = "rank-2"; if(rank === 3) rankClass = "rank-3"; item.className = `list-item ${rankClass}`; // IMAGE FETCHING LOGIC let thumbUrl = ""; if (lvl.video.includes('medal.tv')) { // Placeholder for Medal clips or use a static logo thumbUrl = "https://cdn.medal.tv/assets/img/medal-logo.png"; } else if (lvl.video && lvl.video !== "") { // Get high-quality YouTube thumbnail thumbUrl = `https://img.youtube.com/vi/${lvl.video}/mqdefault.jpg`; } else { // Default image if no video is provided thumbUrl = "https://via.placeholder.com/100x56/1a1d24/ffffff?text=No+Video"; } item.innerHTML = `
#${rank} ${lvl.name} ${lvl.points} PTS
`; item.onclick = () => showLevel(lvl, rank); sidebar.appendChild(item); }); // Automatically display the #1 level on load showLevel(data[0], 1); } catch (e) { console.error("Error loading levels:", e); } } function showLevel(lvl, rank) { document.getElementById('display-title').innerText = `#${rank} ${lvl.name}`; document.getElementById('stat-creator').innerText = lvl.creator; document.getElementById('stat-verifier').innerText = lvl.verifier; document.getElementById('stat-id').innerText = lvl.id; document.getElementById('stat-points').innerText = lvl.points; const v = document.getElementById('display-video'); if(lvl.video.includes('medal.tv')) { const medalId = lvl.video.split('/clips/')[1]?.split('?')[0]; v.src = `https://medal.tv/clip/${medalId}`; } else { v.src = `https://www.youtube.com/embed/${lvl.video || 'dQw4w9WgXcQ'}`; } } start();
