// FETCH AND SORT LEVELS async function loadLevels() { const response = await fetch('levels.json'); let levels = await response.json(); const sidebar = document.getElementById('sidebar-list'); // SORT BY POINTS (Highest to Lowest) levels.sort((a, b) => b.points - a.points); // Render with Rank calculation levels.forEach((lvl, index) => { const displayRank = index + 1; const item = document.createElement('div'); // Add special Top 3 CSS classes automatically based on position let rankClass = ""; if(displayRank === 1) rankClass = "rank-1"; else if(displayRank === 2) rankClass = "rank-2"; else if(displayRank === 3) rankClass = "rank-3"; item.className = `list-item ${rankClass}`; item.innerHTML = `
#${displayRank} ${lvl.name} ${lvl.points} PTS
`; item.onclick = () => displayLevel(lvl, displayRank); sidebar.appendChild(item); }); displayLevel(levels[0], 1); }
